# Squint M0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the existing prototype with a small, installable Gymnasium benchmark that deterministically trains and evaluates detector-refresh policies for tracking-by-detection under a measured compute budget.

**Architecture:** Immutable replay episodes hold cached detections, hidden MOT ground truth, cheap causal scene features, and detector costs. SquintEnv combines one episode, one tracker adapter, and one token bucket; external callable policies choose SKIP or RUN_DETECTOR, while a separate benchmark path writes MOT records and asks TrackEval for HOTA/IDF1. D-FINE-N, ByteTrack, MOT17 preparation, and PPO remain optional reference tooling around a dependency-light core.

**Tech Stack:** Python 3.10-3.13, NumPy 2.x, SciPy, Gymnasium 1.3.x, TrackEval 1.3.x, Hatchling, pytest, Hypothesis, optional roboflow trackers 2.6.0, optional PyTorch 2.11/Transformers 4.57.6, and optional Stable-Baselines3 2.9.0.

# src/squint_rl/tracker.py
from __future__ import annotations

from dataclasses import dataclass
from typing import Protocol, runtime_checkable

import numpy as np
from numpy.typing import NDArray

FloatArray = NDArray[np.float32]
IntArray = NDArray[np.int64]
BoolArray = NDArray[np.bool_]
Observation = dict[str, FloatArray]


def _array(value: object, dtype: np.dtype, shape_tail: tuple[int, ...], name: str):
    result = np.array(value, dtype=dtype, copy=True)
    if result.ndim != len(shape_tail) + 1 or result.shape[1:] != shape_tail:
        raise ValueError(f"{name} must have shape (N, {', '.join(map(str, shape_tail))})")
    if not np.all(np.isfinite(result)):
        raise ValueError(f"{name} must contain only finite values")
    result.setflags(write=False)
    return result


@dataclass(frozen=True, slots=True)
class DetectionBatch:
    boxes_xyxy: FloatArray
    scores: FloatArray
    class_ids: IntArray

    def __post_init__(self) -> None:
        boxes = _array(self.boxes_xyxy, np.dtype(np.float32), (4,), "boxes_xyxy")
        scores = np.array(self.scores, dtype=np.float32, copy=True)
        classes = np.array(self.class_ids, dtype=np.int64, copy=True)
        if scores.shape != (len(boxes),) or classes.shape != (len(boxes),):
            raise ValueError("scores and class_ids must have shape (N,)")
        if not np.all(np.isfinite(scores)) or np.any((scores < 0) | (scores > 1)):
            raise ValueError("scores must be finite values in [0, 1]")
        scores.setflags(write=False)
        classes.setflags(write=False)
        object.__setattr__(self, "boxes_xyxy", boxes)
        object.__setattr__(self, "scores", scores)
        object.__setattr__(self, "class_ids", classes)

    def __len__(self) -> int:
        return len(self.boxes_xyxy)

    @classmethod
    def empty(cls) -> "DetectionBatch":
        return cls(np.empty((0, 4), np.float32), np.empty(0, np.float32), np.empty(0, np.int64))


@dataclass(frozen=True, slots=True)
class GroundTruthBatch:
    boxes_xyxy: FloatArray
    track_ids: IntArray
    class_ids: IntArray
    visibility: FloatArray
    valid: BoolArray
    ignore: BoolArray

    def __post_init__(self) -> None:
        boxes = _array(self.boxes_xyxy, np.dtype(np.float32), (4,), "boxes_xyxy")
        vectors = {
            "track_ids": np.array(self.track_ids, np.int64, copy=True),
            "class_ids": np.array(self.class_ids, np.int64, copy=True),
            "visibility": np.array(self.visibility, np.float32, copy=True),
            "valid": np.array(self.valid, np.bool_, copy=True),
            "ignore": np.array(self.ignore, np.bool_, copy=True),
        }
        if any(value.shape != (len(boxes),) for value in vectors.values()):
            raise ValueError("ground-truth vectors must have shape (N,)")
        if np.any(vectors["valid"] & vectors["ignore"]):
            raise ValueError("ground truth cannot be both valid and ignored")
        if len(np.unique(vectors["track_ids"])) != len(boxes):
            raise ValueError("ground-truth track_ids must be unique within a frame")
        for value in vectors.values():
            value.setflags(write=False)
        object.__setattr__(self, "boxes_xyxy", boxes)
        for name, value in vectors.items():
            object.__setattr__(self, name, value)

    def __len__(self) -> int:
        return len(self.boxes_xyxy)

    @classmethod
    def empty(cls) -> "GroundTruthBatch":
        return cls(
            np.empty((0, 4), np.float32),
            np.empty(0, np.int64),
            np.empty(0, np.int64),
            np.empty(0, np.float32),
            np.empty(0, np.bool_),
            np.empty(0, np.bool_),
        )


@dataclass(frozen=True, slots=True)
class TrackBatch:
    boxes_xyxy: FloatArray
    track_ids: IntArray
    class_ids: IntArray
    scores: FloatArray

    def __post_init__(self) -> None:
        boxes = _array(self.boxes_xyxy, np.dtype(np.float32), (4,), "boxes_xyxy")
        track_ids = np.array(self.track_ids, np.int64, copy=True)
        class_ids = np.array(self.class_ids, np.int64, copy=True)
        scores = np.array(self.scores, np.float32, copy=True)
        if any(value.shape != (len(boxes),) for value in (track_ids, class_ids, scores)):
            raise ValueError("track vectors must have shape (N,)")
        if np.any(track_ids < 0) or len(np.unique(track_ids)) != len(boxes):
            raise ValueError("emitted track_ids must be unique and nonnegative")
        if not np.all(np.isfinite(scores)) or np.any((scores < 0) | (scores > 1)):
            raise ValueError("track scores must be finite values in [0, 1]")
        for value in (track_ids, class_ids, scores):
            value.setflags(write=False)
        object.__setattr__(self, "boxes_xyxy", boxes)
        object.__setattr__(self, "track_ids", track_ids)
        object.__setattr__(self, "class_ids", class_ids)
        object.__setattr__(self, "scores", scores)

    def __len__(self) -> int:
        return len(self.boxes_xyxy)

    @classmethod
    def empty(cls) -> "TrackBatch":
        return cls(
            np.empty((0, 4), np.float32),
            np.empty(0, np.int64),
            np.empty(0, np.int64),
            np.empty(0, np.float32),
        )

~~~python
@dataclass(frozen=True, slots=True)
class TrackerSummary:
    active_tracks: int
    confirmed_tracks: int
    stale_tracks: int
    mean_age_s: float
    mean_motion_px_s: float
    mean_confidence: float

    def __post_init__(self) -> None:
        if min(self.active_tracks, self.confirmed_tracks, self.stale_tracks) < 0:
            raise ValueError("tracker counts must be nonnegative")
        if self.confirmed_tracks > self.active_tracks or self.stale_tracks > self.active_tracks:
            raise ValueError("confirmed and stale counts cannot exceed active tracks")
        if not all(np.isfinite(value) and value >= 0 for value in (self.mean_age_s, self.mean_motion_px_s)):
            raise ValueError("tracker means must be finite and nonnegative")
        if not np.isfinite(self.mean_confidence) or not 0 <= self.mean_confidence <= 1:
            raise ValueError("mean_confidence must be in [0, 1]")

    @classmethod
    def empty(cls) -> "TrackerSummary":
        return cls(0, 0, 0, 0.0, 0.0, 0.0)


@dataclass(frozen=True, slots=True)
class ObservationScales:
    active_tracks: float
    age_s: float
    motion_px_s: float
    time_since_detector_s: float

    def __post_init__(self) -> None:
        values = (self.active_tracks, self.age_s, self.motion_px_s, self.time_since_detector_s)
        if not all(np.isfinite(value) and value > 0 for value in values):
            raise ValueError("observation scales must be finite and positive")


@dataclass(frozen=True, slots=True)
class PolicyContext:
    nominal_rate: float
    source_fps: float
    reserve_ms: float
    seed: int


@runtime_checkable
class Tracker(Protocol):
    def reset(self) -> None:
        pass

    def step(self, detections: DetectionBatch | None, timestamp_s: float) -> TrackBatch:
        pass

    def summary(self) -> TrackerSummary:
        pass

def seal_episode(
    path: str | Path,
    *,
    manifest: Mapping[str, object],
    arrays: Mapping[str, NDArray],
) -> Path:
    destination = Path(path).resolve()
    if destination.exists():
        raise FileExistsError(destination)
    working = destination.parent / f".{destination.name}.{uuid4().hex}.incomplete"
    working.mkdir(parents=True)
    arrays_path = working / "arrays.npz"
    write_deterministic_npz(arrays_path, arrays)
    arrays_hash = sha256(arrays_path.read_bytes()).hexdigest()
    sealed = copy.deepcopy(dict(manifest))
    artifacts = dict(sealed.get("artifacts", {}))
    artifacts["arrays.npz_sha256"] = arrays_hash
    artifacts.pop("content_sha256", None)
    sealed["artifacts"] = artifacts
    normalized = json.dumps(sealed, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()
    artifacts["content_sha256"] = sha256(normalized + arrays_hash.encode()).hexdigest()
    manifest_path = working / "manifest.json"
    manifest_path.write_text(json.dumps(sealed, sort_keys=True, indent=2) + "\n", encoding="utf-8", newline="\n")
    for item in (arrays_path, manifest_path):
        with item.open("rb") as stream:
            os.fsync(stream.fileno())
    os.replace(working, destination)
    return destination


def write_deterministic_npz(path: Path, arrays: Mapping[str, NDArray]) -> None:
    with ZipFile(path, "w", compression=ZIP_STORED) as archive:
        for name in sorted(arrays):
            buffer = BytesIO()
            np.lib.format.write_array(buffer, np.asarray(arrays[name]), allow_pickle=False)
            entry = ZipInfo(f"{name}.npy", date_time=(1980, 1, 1, 0, 0, 0))
            entry.compress_type = ZIP_STORED
            entry.external_attr = 0o600 << 16
            archive.writestr(entry, buffer.getvalue())

# src/squint_rl/budget.py
from dataclasses import dataclass
import math


@dataclass(frozen=True, slots=True)
class BudgetConfig:
    reserve_ms: float
    capacity_ms: float
    refill_ms_per_s: float

    def __post_init__(self) -> None:
        values = (self.reserve_ms, self.capacity_ms, self.refill_ms_per_s)
        if not all(math.isfinite(value) and value > 0 for value in values):
            raise ValueError("budget values must be finite and positive")
        if self.capacity_ms != 2.0 * self.reserve_ms:
            raise ValueError("capacity_ms must equal 2 * reserve_ms")

    @classmethod
    def for_rate(cls, *, reserve_ms: float, source_fps: float, nominal_rate: float) -> "BudgetConfig":
        if not 0.0 < nominal_rate <= 1.0:
            raise ValueError("nominal_rate must be in (0, 1]")
        return cls(reserve_ms, 2.0 * reserve_ms, nominal_rate * source_fps * reserve_ms)


class TokenBucket:
    def __init__(self, config: BudgetConfig) -> None:
        self.config = config
        self.balance_ms = config.reserve_ms
        self._timestamp_s: float | None = None

    @property
    def affordable(self) -> bool:
        return self.balance_ms >= self.config.reserve_ms

    @property
    def normalized_balance(self) -> float:
        return self.balance_ms / self.config.capacity_ms

    def reset(self, *, timestamp_s: float) -> None:
        self.balance_ms = self.config.reserve_ms
        self._timestamp_s = timestamp_s

    def refill(self, *, timestamp_s: float) -> None:
        if self._timestamp_s is None:
            raise RuntimeError("token bucket must be reset before refill")
        elapsed = timestamp_s - self._timestamp_s
        if elapsed < 0:
            raise ValueError("timestamps must be monotonic")
        self.balance_ms = min(
            self.config.capacity_ms,
            self.balance_ms + elapsed * self.config.refill_ms_per_s,
        )
        self._timestamp_s = timestamp_s

    def charge(self, actual_ms: float) -> None:
        if not math.isfinite(actual_ms) or actual_ms < 0:
            raise ValueError("actual_ms must be finite and nonnegative")
        self.balance_ms -= actual_ms
~~~

Debt is finite because every episode latency is validated finite. Do not clamp debt or replace actual cost with reserve_ms.

# src/squint_rl/reward.py
from dataclasses import dataclass, field

import numpy as np
from scipy.optimize import linear_sum_assignment

from squint_rl.tracker import GroundTruthBatch, TrackBatch

IOU_THRESHOLD = 0.5


@dataclass(frozen=True, slots=True)
class MatchCounts:
    matches: int
    false_positives: int
    false_negatives: int
    identity_switches: int
    localization_error: float


def box_iou(left: np.ndarray, right: np.ndarray) -> np.ndarray:
    if len(left) == 0 or len(right) == 0:
        return np.zeros((len(left), len(right)), dtype=np.float32)
    top_left = np.maximum(left[:, None, :2], right[None, :, :2])
    bottom_right = np.minimum(left[:, None, 2:], right[None, :, 2:])
    intersection = np.prod(np.clip(bottom_right - top_left, 0, None), axis=2)
    left_area = np.prod(left[:, 2:] - left[:, :2], axis=1)
    right_area = np.prod(right[:, 2:] - right[:, :2], axis=1)
    union = left_area[:, None] + right_area[None, :] - intersection
    return np.divide(intersection, union, out=np.zeros_like(intersection), where=union > 0)


@dataclass(slots=True)
class RewardState:
    _previous_assignment: dict[int, int] = field(default_factory=dict)

    def reset(self) -> None:
        self._previous_assignment.clear()

    def score(self, ground_truth: GroundTruthBatch, tracks: TrackBatch) -> tuple[float, MatchCounts]:
        valid_mask = ground_truth.valid & ~ground_truth.ignore
        ignored_mask = ground_truth.ignore
        valid_boxes = ground_truth.boxes_xyxy[valid_mask]
        valid_ids = ground_truth.track_ids[valid_mask]
        similarities = box_iou(valid_boxes, tracks.boxes_xyxy)
        rows, cols = linear_sum_assignment(-similarities)
        accepted = similarities[rows, cols] >= IOU_THRESHOLD
        rows, cols = rows[accepted], cols[accepted]

        matched_track_indexes = set(int(index) for index in cols)
        unmatched_tracks = [index for index in range(len(tracks)) if index not in matched_track_indexes]
        ignored_overlap = box_iou(ground_truth.boxes_xyxy[ignored_mask], tracks.boxes_xyxy[unmatched_tracks])
        suppressed = {
            unmatched_tracks[col]
            for col in range(len(unmatched_tracks))
            if len(ignored_overlap) and np.any(ignored_overlap[:, col] >= IOU_THRESHOLD)
        }

        switches = 0
        for row, col in zip(rows, cols, strict=True):
            gt_id = int(valid_ids[row])
            track_id = int(tracks.track_ids[col])
            previous = self._previous_assignment.get(gt_id)
            switches += int(previous is not None and previous != track_id)
            self._previous_assignment[gt_id] = track_id

        counts = MatchCounts(
            matches=len(rows),
            false_positives=len(unmatched_tracks) - len(suppressed),
            false_negatives=len(valid_boxes) - len(rows),
            identity_switches=switches,
            localization_error=float(np.sum(1.0 - similarities[rows, cols])),
        )
        error = counts.false_negatives + counts.false_positives + counts.identity_switches + counts.localization_error
        reward = float(np.clip(1.0 - error / max(1, len(valid_boxes)), -1.0, 1.0))
        return reward, counts
~~~

class EchoTracker:
    def __init__(self) -> None:
        self.last = TrackBatch.empty()
        self.none_calls = 0

    def reset(self) -> None:
        self.last = TrackBatch.empty()
        self.none_calls = 0

    def step(self, detections: DetectionBatch | None, timestamp_s: float) -> TrackBatch:
        if detections is None:
            self.none_calls += 1
            return self.last
        self.last = TrackBatch(
            detections.boxes_xyxy,
            np.arange(1, len(detections) + 1, dtype=np.int64),
            detections.class_ids,
            detections.scores,
        )
        return self.last

    def summary(self) -> TrackerSummary:
        count = len(self.last)
        return TrackerSummary(count, count, 0, 0.0, 0.0, float(np.mean(self.last.scores)) if count else 0.0)

from collections.abc import Callable
from dataclasses import dataclass
from typing import Protocol

import numpy as np

from squint_rl.env import RUN_DETECTOR, SKIP
from squint_rl.tracker import Observation, PolicyContext


class Policy(Protocol):
    def __call__(self, observation: Observation) -> int:
        pass


def reset_policy(policy: Policy, *, seed: int) -> None:
    reset = getattr(policy, "reset", None)
    if callable(reset):
        reset(seed=seed)


def _affordable(observation: Observation) -> bool:
    return bool(observation["compute_budget"][2] >= 0.5)


@dataclass(slots=True)
class GreedyAffordable:
    def __call__(self, observation: Observation) -> int:
        return RUN_DETECTOR if _affordable(observation) else SKIP


@dataclass(slots=True)
class Periodic:
    interval_s: float
    elapsed_scale_s: float

    def __call__(self, observation: Observation) -> int:
        elapsed_s = float(observation["compute_budget"][3]) * self.elapsed_scale_s
        return RUN_DETECTOR if _affordable(observation) and elapsed_s >= self.interval_s else SKIP


@dataclass(slots=True)
class SceneChange:
    threshold: float

    def __call__(self, observation: Observation) -> int:
        changed = float(np.mean(observation["scene_change"])) >= self.threshold
        return RUN_DETECTOR if _affordable(observation) and changed else SKIP


@dataclass(slots=True)
class TrackStaleness:
    threshold: float

    def __call__(self, observation: Observation) -> int:
        stale = float(observation["tracker_state"][2]) >= self.threshold
        return RUN_DETECTOR if _affordable(observation) and stale else SKIP

# src/squint_rl/config.py
JsonValue: TypeAlias = str | int | float | bool | None | list["JsonValue"] | dict[str, "JsonValue"]


class ConfigurationError(ValueError):
    pass


def _strict_keys(value: Mapping[str, object], allowed: set[str], field: str) -> None:
    unknown = sorted(set(value) - allowed)
    if unknown:
        raise ConfigurationError(f"{field}.{unknown[0]} is not supported")


@dataclass(frozen=True, slots=True)
class TrackerSpec:
    factory: str
    parameters: Mapping[str, JsonValue]


@dataclass(frozen=True, slots=True)
class PolicySpec:
    identifier: str
    factory: str
    parameters: Mapping[str, JsonValue]


@dataclass(frozen=True, slots=True)
class BenchmarkConfig:
    source_path: Path
    episodes: tuple[Path, ...]
    output_dir: Path
    budget_rates: tuple[float, ...]
    seed: int
    observation_scales: ObservationScales
    tracker: TrackerSpec
    policies: tuple[PolicySpec, ...]

    @classmethod
    def load(cls, path: str | Path) -> "BenchmarkConfig":
        source = Path(path).resolve()
        with source.open("rb") as stream:
            raw = tomllib.load(stream)
        _strict_keys(
            raw,
            {"schema_version", "episodes", "output_dir", "budget_rates", "seed",
             "observation_scales", "tracker", "policies"},
            "config",
        )
        if raw.get("schema_version") != 1:
            raise ConfigurationError("schema_version must equal 1")
        base = source.parent
        episodes = tuple((base / item).resolve() for item in raw["episodes"])
        missing = next((item for item in episodes if not item.is_dir()), None)
        if missing is not None:
            raise ConfigurationError(f"episodes path does not exist: {missing}")
        rates = tuple(float(item) for item in raw["budget_rates"])
        if rates != tuple(sorted(set(rates))) or any(not 0 < item <= 1 for item in rates):
            raise ConfigurationError("budget_rates must be unique, sorted, and in (0, 1]")
        scale_raw = raw["observation_scales"]
        _strict_keys(
            scale_raw,
            {"active_tracks", "age_s", "motion_px_s", "time_since_detector_s"},
            "observation_scales",
        )
        scales = ObservationScales(**{name: float(value) for name, value in scale_raw.items()})
        tracker_raw = raw["tracker"]
        _strict_keys(tracker_raw, {"factory", "parameters"}, "tracker")
        tracker = TrackerSpec(
            str(tracker_raw["factory"]),
            MappingProxyType(dict(tracker_raw.get("parameters", {}))),
        )
        load_factory(tracker.factory)
        policies = tuple(
            PolicySpec(
                str(item["id"]),
                str(item["factory"]),
                MappingProxyType(dict(item.get("parameters", {}))),
            )
            for item in raw["policies"]
        )
        if len({item.identifier for item in policies}) != len(policies):
            raise ConfigurationError("policies ids must be unique")
        for item in policies:
            load_factory(item.factory)
        return cls(
            source,
            episodes,
            (base / raw["output_dir"]).resolve(),
            rates,
            int(raw["seed"]),
            scales,
            tracker,
            policies,
        )

    def with_policy_override(self, factory: str) -> "BenchmarkConfig":
        load_factory(factory)
        return replace(self, policies=(PolicySpec("external", factory, MappingProxyType({})),))

    def with_output_dir(self, output_dir: str | Path) -> "BenchmarkConfig":
        return replace(self, output_dir=Path(output_dir).resolve())


def load_factory(path: str) -> Callable[..., object]:
    match path.split(":"):
        case ["python", module_name, attribute]:
            module = importlib.import_module(module_name)
            factory = getattr(module, attribute, None)
            if not callable(factory):
                raise ConfigurationError(f"{path} does not resolve to a callable")
            return factory
        case _:
            raise ConfigurationError("factory must use python:module:attribute")

class AtomicRun:
    def __init__(self, destination: str | Path) -> None:
        self.destination = Path(destination).resolve()
        self.working = self.destination.parent / f".{self.destination.name}.{uuid4().hex}.incomplete"

    def __enter__(self) -> Path:
        if self.destination.exists():
            raise FileExistsError(f"completed run already exists: {self.destination}")
        self.destination.parent.mkdir(parents=True, exist_ok=True)
        self.working.mkdir()
        return self.working

    def __exit__(self, exc_type, exc, traceback) -> bool:
        if exc_type is None:
            for item in sorted(path for path in self.working.rglob("*") if path.is_file()):
                with item.open("rb") as stream:
                    os.fsync(stream.fileno())
            os.replace(self.working, self.destination)
        return False

def common_support_areas(
    curves: Mapping[str, Sequence[CurvePoint]],
    *,
    grid_size: int = 101,
) -> CurveAreas:
    ordered = {name: sorted(points, key=lambda point: point.realized_compute) for name, points in curves.items()}
    lower = max(points[0].realized_compute for points in ordered.values())
    upper = min(points[-1].realized_compute for points in ordered.values())
    if lower >= upper:
        raise ValueError("curves have no common measured support")
    grid = np.linspace(lower, upper, grid_size)
    values = {}
    for name, points in ordered.items():
        x = np.array([point.realized_compute for point in points])
        if len(np.unique(x)) != len(x):
            raise ValueError(f"{name} has duplicate compute points")
        y = np.array([point.hota for point in points])
        interpolated = np.interp(grid, x, y)
        values[name] = float(np.trapezoid(interpolated, grid) / (upper - lower))
    return CurveAreas((lower, upper), MappingProxyType(values))

def _metric_summary(raw: Mapping[str, Mapping[str, object]]) -> MetricSummary:
    summary = MetricSummary(

    height = int(sequence_info["imHeight"])
    fps = float(sequence_info["frameRate"])
    extension = sequence_info["imExt"]
    image_dir = source_dir / sequence_info["imDir"]
    image_paths = tuple(image_dir / f"{frame:06d}{extension}" for frame in range(1, length + 1))
    missing = next((path for path in image_paths if not path.is_file()), None)
    if missing is not None:
        raise FileNotFoundError(f"missing source frame: {missing}")

    rows: list[list[tuple[int, int, np.ndarray, float, bool, bool]]] = [[] for _ in range(length)]
    with (source_dir / "gt" / "gt.txt").open(newline="", encoding="utf-8") as stream:
        for line_number, row in enumerate(csv.reader(stream), start=1):
            if len(row) < 9:
                raise ValueError(f"gt.txt line {line_number} must have at least 9 columns")
            frame, identity, x, y, width_box, height_box, mark, class_id, visibility = map(float, row[:9])
            frame_index = int(frame) - 1
            object_class = int(class_id)
            if frame_index not in range(length) or object_class not in range(1, 14):
                raise ValueError(f"gt.txt line {line_number} has an invalid frame or class")
            box = np.array([x - 1, y - 1, x - 1 + width_box, y - 1 + height_box], np.float32)
            if box[0] < 0 or box[1] < 0 or box[2] > width or box[3] > height:
                raise ValueError(f"gt.txt line {line_number} box is outside source dimensions")
            valid = int(mark) == 1 and object_class == 1
            ignore = int(mark) == 1 and object_class in DISTRACTOR_CLASSES
            rows[frame_index].append((int(identity), object_class, box, float(visibility), valid, ignore))

    ground_truth = tuple(_ground_truth_batch(frame_rows) for frame_rows in rows)
    return Mot17Sequence(identifier, source_dir, image_paths, width, height, fps, ground_truth)


def _ground_truth_batch(rows: Sequence[tuple[int, int, np.ndarray, float, bool, bool]]) -> GroundTruthBatch:
    if not rows:
        return GroundTruthBatch.empty()
    identities = [row[0] for row in rows]
    if len(set(identities)) != len(identities):
        raise ValueError("MOT17 frame contains duplicate ground-truth IDs")
    return GroundTruthBatch(
        boxes_xyxy=np.stack([row[2] for row in rows]),
        track_ids=np.array(identities, np.int64),
        class_ids=np.array([row[1] for row in rows], np.int64),
        visibility=np.array([row[3] for row in rows], np.float32),
        valid=np.array([row[4] for row in rows], np.bool_),
        ignore=np.array([row[5] for row in rows], np.bool_),
    )

# src/squint_rl/reference/dfine.py
MODEL_ID = "ustc-community/dfine-nano-coco"
MODEL_REVISION = "066438d3d8f0da137a37b38fdf3368fd4afceced"
THRESHOLD = 0.10


def scene_change_grid(previous: Image.Image | None, current: Image.Image) -> np.ndarray:
    if previous is None:
        return np.zeros((3, 3), dtype=np.float32)
    left = np.asarray(previous.convert("L"), dtype=np.float32)
    right = np.asarray(current.convert("L"), dtype=np.float32)
    difference = np.abs(right - left) / 255.0
    rows = np.array_split(difference, 3, axis=0)
    return np.array([[cell.mean() for cell in np.array_split(row, 3, axis=1)] for row in rows], np.float32)


@dataclass(slots=True)
class DFineDetector:
    processor: object
    model: object
    device: object
    person_label_id: int

    @classmethod
    def load(cls, *, device: str = "cuda", precision: str = "float32") -> "DFineDetector":
        import torch
        from transformers import AutoImageProcessor, AutoModelForObjectDetection

        dtype = {"float32": torch.float32, "float16": torch.float16}[precision]
        processor = AutoImageProcessor.from_pretrained(MODEL_ID, revision=MODEL_REVISION)
        model = AutoModelForObjectDetection.from_pretrained(
            MODEL_ID,
            revision=MODEL_REVISION,
            torch_dtype=dtype,
        ).to(device).eval()
        labels = {int(key): str(value).lower() for key, value in model.config.id2label.items()}
        person_ids = [key for key, value in labels.items() if value == "person"]
        if len(person_ids) != 1:
            raise RuntimeError("D-FINE label map must contain exactly one person class")
        return cls(processor, model, torch.device(device), person_ids[0])

    def predict(self, image: Image.Image) -> tuple[DetectionBatch, float]:
        import torch

        inputs = self.processor(images=image, return_tensors="pt").to(self.device)
        if self.device.type == "cuda":
            torch.cuda.synchronize(self.device)
        started = time.perf_counter()
        with torch.inference_mode():
            outputs = self.model(**inputs)
        if self.device.type == "cuda":
            torch.cuda.synchronize(self.device)
        latency_ms = (time.perf_counter() - started) * 1000.0
        target_sizes = torch.tensor([[image.height, image.width]], device=self.device)
        result = self.processor.post_process_object_detection(
            outputs,
            threshold=THRESHOLD,
            target_sizes=target_sizes,
        )[0]
        keep = result["labels"] == self.person_label_id
        boxes = result["boxes"][keep].detach().cpu().numpy().astype(np.float32)
        scores = result["scores"][keep].detach().cpu().numpy().astype(np.float32)
        classes = np.ones(len(boxes), dtype=np.int64)
        return DetectionBatch(boxes, scores, classes), latency_ms

class FrozenPPOPolicy:
    def __init__(self, checkpoint: str | Path, *, mask_scene_change: bool = False) -> None:
        self.model = PPO.load(checkpoint)
        self.mask_scene_change = mask_scene_change

    def __call__(self, observation: Observation) -> int:
        model_observation = dict(observation)
        if self.mask_scene_change:
            model_observation["scene_change"] = np.zeros_like(observation["scene_change"])
        action, _ = self.model.predict(model_observation, deterministic=True)
        return int(action)


def frozen_policy_factory(
